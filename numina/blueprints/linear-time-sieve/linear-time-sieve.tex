\section{Verified Euler Linear Sieve: correctness and linear-time complexity}

We formalize Euler's linear sieve up to a bound $n \in \mathbb{N}$.
The algorithm outputs an array $\mathrm{lp}$ of least prime factors and a list $\mathrm{primes}$ of discovered primes.
Our main goals are:

\begin{enumerate}
    \item prove that the output is correct;
    \item prove that every composite number is generated exactly once;
    \item deduce a linear upper bound on the number of abstract sieve steps.
\end{enumerate}

We deliberately separate the mathematical complexity proof from any low-level runtime model for Lean arrays.
The complexity theorem is stated in an abstract unit-cost model counting outer-loop iterations, inner-loop visits, and successful writes.

\begin{definition}[Prime list up to $n$]
    \label{def:prime_list_up_to}
    Let $n \in \mathbb{N}$ and let $P$ be a finite list of natural numbers.
    We say that $P$ is a \emph{prime list up to $n$} if:
    \begin{enumerate}
        \item $P$ is strictly increasing;
        \item for every $p \in P$, one has $2 \le p \le n$ and $p$ is prime;
        \item for every prime $p \le n$, one has $p \in P$.
    \end{enumerate}
\end{definition}

\begin{definition}[Least-prime-factor table up to $n$]
    \label{def:lpf_table}
    Let $n \in \mathbb{N}$ and let $\mathrm{lp} : \{0,\dots,n\} \to \mathbb{N}$.
    We say that $\mathrm{lp}$ is a \emph{least-prime-factor table up to $n$} if:
    \begin{enumerate}
        \item $\mathrm{lp}(0)=0$ and $\mathrm{lp}(1)=1$;
        \item for every $m$ with $2 \le m \le n$, the number $\mathrm{lp}(m)$ is prime;
        \item for every $m$ with $2 \le m \le n$, one has $\mathrm{lp}(m)\mid m$;
        \item for every $m$ with $2 \le m \le n$ and every prime $q\mid m$, one has $\mathrm{lp}(m)\le q$.
    \end{enumerate}
\end{definition}

\begin{definition}[Sieve state]
    \label{def:sieve_state}
    A sieve state consists of:
    \begin{enumerate}
        \item a bound $n$;
        \item a current outer index $i$ with $2 \le i \le n+1$;
        \item a partially filled table $\mathrm{lp}$ on $\{0,\dots,n\}$;
        \item a finite list $\mathrm{primes}$;
        \item a cost counter $\mathrm{cost}\in\mathbb{N}$.
    \end{enumerate}
\end{definition}

\begin{definition}[Abstract sieve step count]
    \label{def:cost_model}
    We use the following abstract cost model.
    \begin{enumerate}
        \item Advancing the outer loop from $i$ to $i+1$ costs $1$.
        \item Visiting one prime $p$ in the inner loop for the current index $i$ costs $1$.
        \item Writing $\mathrm{lp}(i\cdot p):=p$ has no extra cost beyond the visit itself.
    \end{enumerate}
    Thus the total cost is
    \[
        \mathrm{cost}
        =
        (\text{number of outer-loop iterations})
        +
        (\text{number of inner-loop visits}).
    \]
\end{definition}

\begin{definition}[Linear sieve decomposition]
    \label{def:linear_decomp}
    Let $m\ge 2$.
    A \emph{linear sieve decomposition} of $m$ is a pair $(a,p)$ such that:
    \begin{enumerate}
        \item $p$ is prime;
        \item $m = a p$;
        \item $p$ is the least prime factor of $m$;
        \item $p \le \mathrm{lp}(a)$ whenever $\mathrm{lp}(a)$ is defined.
    \end{enumerate}
\end{definition}

\begin{definition}[Outer-loop invariant]
    \label{def:outer_invariant}
    \uses{def:prime_list_up_to, def:lpf_table, def:sieve_state}
    Let $S$ be a sieve state with current index $i$.
    We say that $S$ satisfies the \emph{outer-loop invariant at $i$} if:
    \begin{enumerate}
        \item for every $m$ with $2\le m < i$, the value $\mathrm{lp}(m)$ is already correct;
        \item the list $\mathrm{primes}$ is exactly the strictly increasing list of all primes $< i$;
        \item for every prime $p < i$, one has $\mathrm{lp}(p)=p$;
        \item every composite $m < i$ has already been generated by the sieve rule.
    \end{enumerate}
\end{definition}

\begin{lemma}[Existence of a least prime factor]
    \label{lem:lpf_exists}
    For every integer $m\ge 2$, there exists a prime $p$ such that $p\mid m$ and
    for every prime $q\mid m$ one has $p\le q$.
\end{lemma}
\begin{proof}
    By well-ordering of the naturals, the set of prime divisors of $m$ is nonempty (since $m \ge 2$) and has a minimum element.
\end{proof}

\begin{lemma}[No smaller prime divisor implies primality]
    \label{lem:no_small_divisor_prime}
    \uses{lem:lpf_exists}
    Let $m\ge 2$.
    If there is no prime $p < m$ with $p\mid m$, then $m$ is prime.
\end{lemma}
\begin{proof}
    \uses{lem:lpf_exists}
    By \ref{lem:lpf_exists}, $m$ has a least prime factor $p$. Since no prime less than $m$ divides $m$, we must have $p = m$, so $m$ is prime.
\end{proof}

\begin{lemma}[A smaller prime divisor of $a$ is also a divisor of $ap$]
    \label{lem:divisor_lifts_to_product}
    Let $a,p,q\in\mathbb{N}$.
    If $q\mid a$, then $q\mid ap$.
\end{lemma}
\begin{proof}
    If $q \mid a$, then $a = qk$ for some $k$, so $ap = qkp$ and $q \mid ap$.
\end{proof}

\begin{lemma}[Least-prime-factor monotonicity along the decomposition]
    \label{lem:lpf_monotone_along_decomp}
    \uses{lem:divisor_lifts_to_product}
    Let $m\ge 2$, and let $p$ be the least prime factor of $m$.
    Suppose $m=ap$.
    Then every prime divisor $q$ of $a$ satisfies $p\le q$.
    In particular, if $a\ge 2$, then $p \le \mathrm{lp}(a)$.
\end{lemma}
\begin{proof}
    \uses{lem:divisor_lifts_to_product}
    If $q$ is a prime divisor of $a$, then by \ref{lem:divisor_lifts_to_product}, $q \mid ap = m$. Since $p$ is the least prime factor of $m$, we have $p \le q$.
\end{proof}

\begin{lemma}[Correctness of a write]
    \label{lem:write_correct}
    \uses{lem:lpf_monotone_along_decomp}
    Let $i\ge 2$ and let $p$ be a prime such that $p \le \mathrm{lp}(i)$.
    Then $p$ is the least prime factor of $ip$.
\end{lemma}
\begin{proof}
    \uses{lem:lpf_monotone_along_decomp}
    Since $p$ is prime and $p \mid ip$, it suffices to show no prime $q < p$ divides $ip$. Any such $q$ would divide $i$ (since $q < p$ and $p$ is prime), contradicting $p \le \mathrm{lp}(i)$.
\end{proof}

\begin{lemma}[A composite is generated before its own outer iteration]
    \label{lem:composite_generated_early}
    \uses{lem:lpf_monotone_along_decomp, lem:write_correct}
    Let $m\ge 2$ be composite, and let $p$ be its least prime factor.
    Write $m = ap$.
    Then $a < m$, and when the sieve processes the outer index $a$, the inner loop reaches the prime $p$ and writes $\mathrm{lp}(m):=p$.
\end{lemma}
\begin{proof}
    \uses{lem:lpf_monotone_along_decomp, lem:write_correct}
    Since $m$ is composite, $p < m$, so $a = m/p \ge 2$ and $a < m$. By \ref{lem:lpf_monotone_along_decomp}, $p \le \mathrm{lp}(a)$, so the inner loop visits $p$. By \ref{lem:write_correct}, the write is correct.
\end{proof}

\begin{lemma}[If $\mathrm{lp}(i)=0$, then $i$ is prime]
    \label{lem:lp_zero_implies_prime}
    \uses{lem:no_small_divisor_prime, lem:composite_generated_early}
    Let $i\ge 2$.
    Assume the outer-loop invariant holds at index $i$, and suppose $\mathrm{lp}(i)=0$ at that moment.
    Then $i$ is prime.
\end{lemma}
\begin{proof}
    \uses{lem:no_small_divisor_prime, lem:composite_generated_early}
    If $i$ were composite, by \ref{lem:composite_generated_early} the sieve would have already set $\mathrm{lp}(i) \ne 0$. So no prime $p < i$ divides $i$, and by \ref{lem:no_small_divisor_prime}, $i$ is prime.
\end{proof}

\begin{lemma}[Appending a newly found prime preserves the prime list property]
    \label{lem:append_prime_preserves_prime_list}
    \uses{def:prime_list_up_to, lem:lp_zero_implies_prime}
    Suppose the outer-loop invariant holds at index $i$ and $\mathrm{lp}(i)=0$.
    Then appending $i$ to $\mathrm{primes}$ preserves the statement that $\mathrm{primes}$ is the strictly increasing list of all primes $\le i$.
\end{lemma}
\begin{proof}
    \uses{lem:lp_zero_implies_prime}
    By \ref{lem:lp_zero_implies_prime}, $i$ is prime. Since all primes $< i$ are already in the list and the list is sorted, appending $i$ maintains both properties.
\end{proof}

\begin{lemma}[Inner-loop writes are correct]
    \label{lem:inner_loop_correct}
    \uses{lem:write_correct}
    Fix an outer index $i$ satisfying the invariant.
    For every visited prime $p$ with $ip\le n$ and $p\le \mathrm{lp}(i)$, the assignment $\mathrm{lp}(ip):=p$ is correct.
\end{lemma}
\begin{proof}
    \uses{lem:write_correct}
    Direct application of \ref{lem:write_correct}.
\end{proof}

\begin{lemma}[Preservation of the outer-loop invariant]
    \label{lem:outer_invariant_step}
    \uses{def:outer_invariant, lem:append_prime_preserves_prime_list, lem:inner_loop_correct}
    Suppose the outer-loop invariant holds at index $i$ with $2\le i\le n$.
    After processing index $i$ by the linear sieve rule, the resulting state satisfies the outer-loop invariant at index $i+1$.
\end{lemma}
\begin{proof}
    \uses{lem:append_prime_preserves_prime_list, lem:inner_loop_correct}
    The inner loop correctly fills $\mathrm{lp}$ for composites of the form $ip$ by \ref{lem:inner_loop_correct}. If $i$ is a newly discovered prime, appending it is justified by \ref{lem:append_prime_preserves_prime_list}. All four conditions of the invariant at $i+1$ follow.
\end{proof}

\begin{theorem}[Correctness of the least-prime-factor table]
    \label{thm:lpf_correct}
    \uses{def:lpf_table, def:outer_invariant, lem:outer_invariant_step}
    After the sieve terminates, the output table $\mathrm{lp}$ is a least-prime-factor table up to $n$.
\end{theorem}
\begin{proof}
    \uses{lem:outer_invariant_step}
    By induction on the outer index $i$ from $2$ to $n+1$, using \ref{lem:outer_invariant_step} at each step. The invariant at $n+1$ implies every entry in $\mathrm{lp}$ is correct.
\end{proof}

\begin{theorem}[Correctness of the prime list]
    \label{thm:prime_list_correct}
    \uses{def:prime_list_up_to, def:outer_invariant, lem:outer_invariant_step}
    After the sieve terminates, the output list $\mathrm{primes}$ is a prime list up to $n$.
\end{theorem}
\begin{proof}
    \uses{lem:outer_invariant_step}
    By the same induction as for \ref{thm:lpf_correct}. The invariant at $n+1$ ensures the prime list contains all primes up to $n$ and is strictly increasing.
\end{proof}

\begin{lemma}[Existence of linear sieve decomposition]
    \label{lem:linear_decomp_exists}
    \uses{lem:lpf_exists, lem:lpf_monotone_along_decomp, def:linear_decomp}
    Every composite integer $m\le n$ admits a linear sieve decomposition.
\end{lemma}
\begin{proof}
    \uses{lem:lpf_exists, lem:lpf_monotone_along_decomp}
    Let $p$ be the least prime factor of $m$ (exists by \ref{lem:lpf_exists}) and set $a = m/p$. Then $(a,p)$ satisfies all conditions by \ref{lem:lpf_monotone_along_decomp}.
\end{proof}

\begin{lemma}[Uniqueness of linear sieve decomposition]
    \label{lem:linear_decomp_unique}
    \uses{def:linear_decomp}
    Let $m$ be composite.
    Then there is at most one linear sieve decomposition $(a,p)$ of $m$.
\end{lemma}
\begin{proof}
    \uses{def:linear_decomp}
    If $(a_1, p_1)$ and $(a_2, p_2)$ are both decompositions, then both $p_1$ and $p_2$ are the least prime factor of $m$, so $p_1 = p_2$. Then $a_1 = m/p_1 = m/p_2 = a_2$.
\end{proof}

\begin{theorem}[Every composite is generated exactly once]
    \label{thm:every_composite_once}
    \uses{lem:linear_decomp_exists, lem:linear_decomp_unique}
    During the execution of the linear sieve up to $n$, each composite number $m\le n$ is written exactly once in the table $\mathrm{lp}$.
\end{theorem}
\begin{proof}
    \uses{lem:linear_decomp_exists, lem:linear_decomp_unique}
    Existence of a write follows from \ref{lem:linear_decomp_exists}; uniqueness follows from \ref{lem:linear_decomp_unique}.
\end{proof}

\begin{lemma}[Bound on successful inner-loop visits]
    \label{lem:successful_visits_bound}
    \uses{thm:every_composite_once}
    The number of inner-loop visits that successfully write a value $\mathrm{lp}(m)$ is at most the number of composite integers in $\{2,\dots,n\}$.
    In particular, it is at most $n-1$.
\end{lemma}
\begin{proof}
    \uses{thm:every_composite_once}
    Each successful write corresponds to a unique composite by \ref{thm:every_composite_once}, and there are at most $n-1$ composites in $\{2,\dots,n\}$.
\end{proof}

\begin{lemma}[At most one extra terminating inner-loop visit per outer index]
    \label{lem:extra_visit_bound}
    For each outer index $i$, the inner loop performs at most one additional visit that does not produce a successful write:
    namely, the first visit at which either $ip>n$ or $p>\mathrm{lp}(i)$.
\end{lemma}
\begin{proof}
    The inner loop terminates at the first failing condition, producing at most one non-writing visit per outer index.
\end{proof}

\begin{lemma}[Bound on all inner-loop visits]
    \label{lem:all_inner_visits_bound}
    \uses{lem:successful_visits_bound, lem:extra_visit_bound}
    The total number of inner-loop visits is at most
    \[
        (\text{number of composites }\le n) + (n-1).
    \]
    In particular, it is at most $2n-2$.
\end{lemma}
\begin{proof}
    \uses{lem:successful_visits_bound, lem:extra_visit_bound}
    Add the bound from \ref{lem:successful_visits_bound} (at most $n-1$ successful visits) and from \ref{lem:extra_visit_bound} (at most $n-1$ extra visits).
\end{proof}

\begin{lemma}[Bound on outer-loop iterations]
    \label{lem:outer_iterations_bound}
    The outer loop runs exactly once for each $i\in\{2,\dots,n\}$.
    Hence the number of outer-loop iterations is $n-1$.
\end{lemma}
\begin{proof}
    The outer loop iterates $i = 2, 3, \dots, n$, which is exactly $n-1$ iterations.
\end{proof}

\begin{theorem}[Linear-time bound in the abstract cost model]
    \label{thm:linear_time}
    \uses{def:cost_model, lem:all_inner_visits_bound, lem:outer_iterations_bound}
    The total abstract cost of the linear sieve up to $n$ is at most
    \[
        (n-1) + (2n-2) = 3n-3.
    \]
    In particular, the sieve runs in linear time:
    \[
        \mathrm{cost} = O(n).
    \]
\end{theorem}
\begin{proof}
    \uses{lem:all_inner_visits_bound, lem:outer_iterations_bound}
    Sum the outer-loop cost ($n-1$ by \ref{lem:outer_iterations_bound}) and the inner-loop cost (at most $2n-2$ by \ref{lem:all_inner_visits_bound}).
\end{proof}

\begin{corollary}[Verified linear sieve]
    \label{cor:verified_linear_sieve}
    \uses{thm:lpf_correct, thm:prime_list_correct, thm:linear_time}
    The formalized Euler sieve computes the correct least-prime-factor table and the correct list of primes up to $n$, and it satisfies a linear-time bound in the abstract unit-cost model.
\end{corollary}
\begin{proof}
    \uses{thm:lpf_correct, thm:prime_list_correct, thm:linear_time}
    Combine \ref{thm:lpf_correct}, \ref{thm:prime_list_correct}, and \ref{thm:linear_time}.
\end{proof}
